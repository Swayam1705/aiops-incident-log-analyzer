1. Check the pods:
   kubectl -n {{ .Release.Namespace }} get pods -l app.kubernetes.io/instance={{ .Release.Name }}

2. Port-forward the UI:
   kubectl -n {{ .Release.Namespace }} port-forward svc/{{ include "aiops.fullname" . }} 8000:{{ .Values.service.port }}
   then open http://localhost:8000/ui

3. Inject a failure and analyze:
   python scripts/inject_errors.py http-500-burst
   python scripts/run_analyzer.py
